def append_list(path,key,lines):
    t=open(path,encoding='utf8').read()
    i=t.index(f'  {key}:\n')+len(f'  {key}:\n'); j=i
    while t.startswith('  - {',j): j=t.index('\n',j)+1
    open(path,'w',encoding='utf8').write(t[:j]+''.join(lines)+t[j:])
