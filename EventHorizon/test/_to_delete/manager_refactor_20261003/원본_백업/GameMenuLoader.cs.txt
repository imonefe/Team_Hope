using UnityEngine;

/// <summary>
/// 게임 시작 시 GameMenuCanvas 프리팹을 한 번 생성해두고(평소엔 비활성 상태),
/// ESC 입력을 감지해서 필요할 때만 켜주는 역할을 한다.
///
/// GameMenuCanvas 자체는 켜지는 순간(OnEnable) 자동으로 설정 패널을 띄우고 게임을 정지시키며,
/// 닫힐 때(Resume) 스스로 다시 비활성화된다 — 그래서 이 스크립트는 "한 번 생성해두는 것"과
/// "평소엔 꺼둔 채로 있다가 ESC 눌리면 켜주는 것"만 담당하면 된다.
///
/// 씬에 따로 오브젝트를 더 두지 않아도 되도록, 프리팹을 씬에 미리 배치해두는 대신
/// 이 스크립트가 시작할 때 코드로 한 번 Instantiate 해두는 방식을 쓴다.
/// </summary>
public class GameMenuLoader : MonoBehaviour
{
    [Tooltip("게임 시작 시 한 번 생성해둘 설정 메뉴 프리팹")]
    public GameObject gameMenuCanvasPrefab;

    GameObject instance;

    void Awake()
    {
        // 씬 안에 이미 설정 메뉴가 있다면(수동으로 배치해둔 경우 등) 중복 생성하지 않고 그걸 그대로 씀
        GameMenuController existing = FindObjectOfType<GameMenuController>();
        if (existing != null)
        {
            instance = existing.gameObject;
            return;
        }

        if (gameMenuCanvasPrefab == null)
        {
            Debug.LogWarning("[GameMenuLoader] gameMenuCanvasPrefab이 비어있습니다.");
            return;
        }

        instance = Instantiate(gameMenuCanvasPrefab);

        // 프리팹의 루트 오브젝트는 기본적으로 활성 상태로 생성되기 때문에(그 순간 OnEnable도 같이
        // 실행되어 설정 패널이 잠깐 열림), 렌더링이 일어나기 전인 이 시점에 바로 꺼서
        // 평소 상태(비활성)로 되돌린다. 같은 프레임 안에서 처리되므로 화면 깜빡임은 없다.
        instance.SetActive(false);
    }

    void Update()
    {
        if (instance == null) return;
        if (instance.activeSelf) return; // 이미 열려있으면 GameMenuController가 알아서 처리하므로 무시

        if (!Input.GetKeyDown(KeyCode.Escape)) return;

        // 인벤토리가 열려있는 상태에서 ESC를 누르면 설정 메뉴를 열지 않고 인벤토리부터 닫는다.
        if (Inventory.Instance != null && Inventory.Instance.IsOpen)
        {
            Inventory.Instance.Close();
            return;
        }

        instance.SetActive(true);
    }
}
